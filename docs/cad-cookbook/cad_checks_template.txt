"""jobs/<id>/cad_checks.py: prove the critical dimensions on the built geometry.

Run from the project root: python3 jobs/<id>/cad_checks.py
Exit code 0 = every check passed. Copy this, then edit the CHECKS section.
"""
import runpy
import sys
from pathlib import Path

JOB = Path(__file__).resolve().parent
sys.path.insert(0, str(JOB.parents[1] / "tools"))
from cadlib import bbox, measure_holes  # noqa: E402

part = runpy.run_path(str(JOB / "model.py"), run_name="model")["build"]()
size = bbox(part)["size"]
holes = measure_holes(part)
failures = []


def check(name, ok, detail):
    print(f"{'PASS' if ok else 'FAIL'}  {name}: {detail}")
    if not ok:
        failures.append(name)


# ---- CHECKS: one line per critical dimension in design_brief.md -----------------
check("overall size", all(abs(a - b) < 0.01 for a, b in zip(size, [40.0, 40.0, 20.0])), size)
check("two M4 clearance holes at 4.5 + HOLE_COMP", sorted(h["diameter"] for h in holes) == [4.7, 4.7],
      [h["diameter"] for h in holes])
# ---------------------------------------------------------------------------------

sys.exit(1 if failures else 0)
